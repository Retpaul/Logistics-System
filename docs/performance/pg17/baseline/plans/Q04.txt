FROM "Order" o
WHERE o."tenantId" = '76369a2d-016f-7bbd-d447-684b65cb44af'
  AND o."createdAt" >= date_trunc('day', now() AT TIME ZONE 'Africa/Lagos') AT TIME ZONE 'Africa/Lagos'
ORDER BY o."createdAt" DESC
LIMIT 25;
